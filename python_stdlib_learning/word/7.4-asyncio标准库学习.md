# Python 3.14 `asyncio` 标准库完全学习指南

## 一、概念：异步编程与 `asyncio` 基础
### 1. 同步 vs 异步：为什么需要异步？
- **同步编程**：代码按顺序执行，遇到 I/O 操作（如网络请求、文件读写）会**阻塞等待**，直到操作完成才继续。
- **异步编程**：遇到 I/O 操作时**不阻塞**，转而执行其他任务，等 I/O 完成后再回来处理结果。

### 2. `asyncio` 的核心价值
`asyncio` 是 Python 标准库中用于**异步 I/O 编程**的框架，核心优势：
- **高并发**：单线程内通过协程切换实现并发，无需线程/进程创建开销；
- **低资源占用**：协程是“用户态轻量级线程”，内存占用远低于线程；
- **适用场景**：网络爬虫、Web 服务器、数据库连接等 **I/O 密集型任务**。

### 3. 核心概念
| 概念         | 说明                                  |
|--------------|---------------------------------------|
| **协程（Coroutine）** | 用 `async def` 定义的函数，可在执行中暂停（`await`）并切换到其他协程 |
| **事件循环（Event Loop）** | `asyncio` 的核心，负责调度协程、处理 I/O 事件 |
| **Task** | 协程的“包装器”，将协程提交到事件循环并发执行 |
| **Awaitable** | 可被 `await` 的对象（协程、Task、Future 等） |


## 二、基础部分：核心用法
### 1. 协程的定义与运行
用 `async def` 定义协程，用 `await` 暂停执行，用 `asyncio.run()` 启动事件循环：
```python
import asyncio

async def greet(name):
    """简单协程：模拟 I/O 操作"""
    print(f"Hello, {name}!")
    await asyncio.sleep(1)  # 模拟 I/O 等待（必须用 asyncio.sleep，不能用 time.sleep）
    print(f"Goodbye, {name}!")
    return f"Greeting to {name} done"

# 运行协程（asyncio.run() 是 Python 3.7+ 推荐的入口点）
result = asyncio.run(greet("Alice"))
print("协程返回结果:", result)
```

### 2. 并发执行协程：`Task` 与 `gather()`
#### 方式 1：用 `asyncio.create_task()` 创建 Task
`Task` 会将协程提交到事件循环，实现**并发执行**：
```python
async def task(name, delay):
    """模拟耗时 I/O 任务"""
    print(f"[Task {name}] 开始执行")
    await asyncio.sleep(delay)
    print(f"[Task {name}] 执行完成（耗时 {delay}s）")
    return f"Result-{name}"

async def main():
    # 创建 Task（立即提交到事件循环，不阻塞）
    task1 = asyncio.create_task(task("A", 2))
    task2 = asyncio.create_task(task("B", 1))
    
    # 等待 Task 完成（await 会暂停当前协程，直到 Task 完成）
    result1 = await task1
    result2 = await task2
    print("所有 Task 结果:", result1, result2)

asyncio.run(main())
```

#### 方式 2：用 `asyncio.gather()` 批量并发
`gather()` 可同时运行多个协程/Task，按**提交顺序**返回结果：
```python
async def main():
    # 批量并发执行，按顺序返回结果
    results = await asyncio.gather(
        task("A", 2),
        task("B", 1),
        task("C", 3)
    )
    print("gather 结果:", results)  # ['Result-A', 'Result-B', 'Result-C']

asyncio.run(main())
```

### 3. `Future`：底层异步结果对象
`Future` 是 `Task` 的基类，代表“未来会完成的操作”，通常无需直接使用，但需了解其核心方法：
```python
async def main():
    # 创建 Future（通常由框架内部创建，这里仅作演示）
    future = asyncio.Future()
    
    # 模拟异步操作完成后设置结果
    async def set_result():
        await asyncio.sleep(1)
        future.set_result("Future 完成了！")
    
    # 并发执行设置结果的协程
    asyncio.create_task(set_result())
    
    # 等待 Future 完成
    result = await future
    print("Future 结果:", result)

asyncio.run(main())
```


## 三、进阶部分：深度技巧
### 1. 异步上下文管理器：`async with`
用于需要“异步初始化/清理”的资源（如网络连接、数据库连接），需实现 `__aenter__` 和 `__aexit__` 方法：
```python
# 示例：用 aiohttp（第三方库）的异步上下文管理器管理 HTTP 会话
# 安装：pip install aiohttp
import aiohttp

async def fetch_url(session, url):
    """异步发送 HTTP 请求"""
    async with session.get(url) as response:  # async with 管理响应资源
        return await response.text()

async def main():
    async with aiohttp.ClientSession() as session:  # async with 管理会话资源
        html = await fetch_url(session, "https://www.example.com")
        print("网页内容长度:", len(html))

asyncio.run(main())
```

### 2. 异步迭代器：`async for`
用于异步生成数据序列，需实现 `__aiter__` 和 `__anext__` 方法，或用 `async def` + `yield` 定义异步生成器：
```python
async def async_counter(n):
    """异步生成器：每隔 0.5s 生成一个数字"""
    for i in range(n):
        await asyncio.sleep(0.5)
        yield i  # 异步生成数据

async def main():
    # 用 async for 遍历异步生成器
    async for num in async_counter(5):
        print("收到数字:", num)

asyncio.run(main())
```

### 3. 同步原语：控制协程并发
协程虽在单线程内运行，但共享资源时仍需同步（如限制并发数、保护共享变量）。

#### 3.1 `Semaphore`：限制并发数
```python
async def limited_task(sem, name):
    """用 Semaphore 限制同时执行的协程数"""
    async with sem:  # 获取信号量，超过限制则等待
        print(f"[Task {name}] 获得执行权")
        await asyncio.sleep(1)
        print(f"[Task {name}] 释放执行权")

async def main():
    sem = asyncio.Semaphore(2)  # 最多允许 2 个协程同时执行
    tasks = [asyncio.create_task(limited_task(sem, f"Task-{i}")) for i in range(5)]
    await asyncio.gather(*tasks)

asyncio.run(main())
```

#### 3.2 `Lock`：保护共享资源
```python
async def increment(lock, shared_value):
    """用 Lock 保护共享变量"""
    for _ in range(10000):
        async with lock:  # 获取锁，确保同一时间只有一个协程修改
            shared_value[0] += 1

async def main():
    lock = asyncio.Lock()
    shared_value = [0]  # 用列表模拟可变共享变量
    
    # 并发执行 2 个累加协程
    tasks = [asyncio.create_task(increment(lock, shared_value)) for _ in range(2)]
    await asyncio.gather(*tasks)
    
    print("共享变量最终值:", shared_value[0])  # 无锁时结果可能小于 20000

asyncio.run(main())
```

### 4. 协程间通信：`asyncio.Queue`
`Queue` 是协程安全的队列，用于生产者-消费者场景：
```python
async def producer(queue):
    """生产者：向队列中放入数据"""
    for i in range(5):
        await asyncio.sleep(0.5)
        await queue.put(i)  # 队列满时会阻塞
        print(f"[生产者] 放入数据: {i}")
    await queue.put(None)  # 用 None 作为结束标记

async def consumer(queue):
    """消费者：从队列中取出数据"""
    while True:
        item = await queue.get()  # 队列空时会阻塞
        if item is None:
            break
        await asyncio.sleep(1)
        print(f"[消费者] 取出数据: {item}")
        queue.task_done()  # 标记任务完成

async def main():
    queue = asyncio.Queue(maxsize=2)  # 队列最大容量为 2
    prod_task = asyncio.create_task(producer(queue))
    cons_task = asyncio.create_task(consumer(queue))
    
    await prod_task  # 等待生产者完成
    await queue.join()  # 等待队列中所有任务被处理
    await cons_task  # 等待消费者退出

asyncio.run(main())
```

### 5. 异常处理进阶
#### 5.1 `Task` 中的异常捕获
`Task` 的异常会在 `await` 时抛出，需手动捕获：
```python
async def error_task():
    raise ValueError("任务出错了！")

async def main():
    task = asyncio.create_task(error_task())
    try:
        await task
    except ValueError as e:
        print(f"捕获到 Task 异常: {e}")

asyncio.run(main())
```

#### 5.2 `gather()` 中的异常处理
用 `return_exceptions=True` 可将异常作为结果返回，避免中断其他任务：
```python
async def maybe_error(x):
    if x == 2:
        raise ValueError(f"x={x} 出错了")
    return x * x

async def main():
    results = await asyncio.gather(
        maybe_error(1),
        maybe_error(2),
        maybe_error(3),
        return_exceptions=True  # 异常作为结果返回
    )
    print("gather 结果:", results)  # [1, ValueError('x=2 出错了'), 9]

asyncio.run(main())
```


## 四、实战案例：高并发异步网页爬虫
### 需求
用 `asyncio` + `aiohttp` 实现一个高并发网页爬虫，要求：
1. 异步下载多个网页；
2. 用 `Semaphore` 限制并发数（避免被封 IP）；
3. 保存网页内容到本地文件。

### 代码实现
```python
import asyncio
import aiohttp
import os
from typing import List

# ---------------------- 配置 ----------------------
URLS = [
    "https://www.example.com",
    "https://www.python.org",
    "https://www.github.com",
    "https://www.stackoverflow.com",
    "https://www.baidu.com",
]
SAVE_DIR = "downloaded_pages"
os.makedirs(SAVE_DIR, exist_ok=True)
MAX_CONCURRENT = 3  # 最大并发数

# ---------------------- 核心逻辑 ----------------------
async def download_page(sem: asyncio.Semaphore, session: aiohttp.ClientSession, url: str) -> str:
    """异步下载单个网页"""
    async with sem:  # 限制并发
        try:
            print(f"[下载] 开始: {url}")
            async with session.get(url, timeout=10) as response:
                response.raise_for_status()  # 检查 HTTP 错误
                content = await response.text()
                print(f"[下载] 完成: {url} (长度: {len(content)})")
                return content
        except Exception as e:
            print(f"[下载] 失败: {url}, 错误: {e}")
            return ""

async def save_page(url: str, content: str) -> None:
    """异步保存网页到本地（这里用同步文件操作模拟，实际可用 aiofiles）"""
    if not content:
        return
    filename = f"page_{hash(url) % 10000}.html"
    save_path = os.path.join(SAVE_DIR, filename)
    with open(save_path, "w", encoding="utf-8") as f:
        f.write(content)
    print(f"[保存] 完成: {save_path}")

async def main(urls: List[str]):
    """主流程：并发下载 + 保存"""
    sem = asyncio.Semaphore(MAX_CONCURRENT)
    async with aiohttp.ClientSession() as session:
        # 1. 并发下载所有网页
        tasks = [asyncio.create_task(download_page(sem, session, url)) for url in urls]
        contents = await asyncio.gather(*tasks)
        
        # 2. 保存网页（这里简化为顺序保存，实际也可异步化）
        for url, content in zip(urls, contents):
            await save_page(url, content)

# ---------------------- 启动 ----------------------
if __name__ == "__main__":
    print(f"=== 开始爬虫（并发数: {MAX_CONCURRENT}）===")
    asyncio.run(main(URLS))
    print("=== 爬虫完成 ===")
```


## 五、总结
1. **核心选择**：`asyncio` 是 **I/O 密集型高并发任务**的首选（如网络爬虫、Web 服务器），CPU 密集型任务建议用 `multiprocessing`。
2. **核心用法**：
   - 协程用 `async def` 定义，`await` 暂停，`asyncio.run()` 启动；
   - 并发执行用 `create_task()` 或 `gather()`；
   - 资源管理用 `async with`，异步迭代用 `async for`；
   - 同步用 `Semaphore`/`Lock`，通信用 `Queue`。
3. **避坑提醒**：
   - 不要在协程中调用**阻塞函数**（如 `time.sleep()`、同步文件读写），要用异步版本（如 `asyncio.sleep()`、`aiofiles`）；
   - 异常需在 `await` 时捕获，或用 `gather(return_exceptions=True)` 处理。
